72// Fisher-Yates shuffle
73void Deck::shuffle() {
74 for (size_t i = 0; i < cards_.size(); ++i) {
75 std::uniform_int_distribution<size_t> dist(i, cards_.size() - 1);
76 size_t swapIdx = dist(rng_);
77 std::swap(cards_[i], cards_[swapIdx]);
78 }
79}
